on_game_start = {
	on_actions = {
		villains_rasputin_camp_purpose_on_game_start
		villains_mutaguchi_house_aspiration_on_game_start
	}
}

villains_rasputin_camp_purpose_on_game_start = {
	effect = {
		character:reincarnated_rasputin_867 ?= {
			if = {
				limit = {
					is_alive = yes
					has_government = landless_adventurer_government
				}
				add_realm_law_skip_effects = camp_purpose_scholars
				remove_camp_purpose_flavourisation_variables_effect = yes
				set_variable = flavourise_camp_purpose_scholars
			}
		}
	}
}

villains_mutaguchi_house_aspiration_on_game_start = {
	effect = {
		character:reincarnated_mutaguchi_867 ?= {
			if = {
				limit = {
					is_alive = yes
					has_tgp_dlc_trigger = yes
					government_is_japanese_trigger = yes
				}
				house ?= {
					if = {
						limit = {
							NOT = {
								AND = {
									has_house_aspiration_parameter = aspiration_strength
									has_house_aspiration_parameter = aspiration_level_1
								}
							}
						}
						set_house_aspiration = {
							type = strength
							level = 1
						}
					}
				}
			}
		}
	}
}

on_game_start_after_lobby = {

	events = {

		villains_relics.0001

	}

}

on_trait_gained = {
	on_actions = {
		villains_rasputin_incapable_immunity_on_action
	}
}

villains_rasputin_incapable_immunity_on_action = {
	effect = {
		if = {
			limit = {
				scope:trait = trait:incapable
				has_character_modifier = rasputin_herbal_contraception_modifier
			}
			remove_trait = incapable
			if = {
				limit = { has_character_flag = had_event_harm_0002 }
				remove_character_flag = had_event_harm_0002
			}
			if = {
				limit = { has_character_flag = had_event_harm_0003 }
				remove_character_flag = had_event_harm_0003
			}
			if = {
				limit = { has_trait = withering_mind }
				add_trait_xp = {
					trait = withering_mind
					track = withering_mind
					value = -50
				}
			}
			if = {
				limit = { has_trait = fragile_bones }
				add_trait_xp = {
					trait = fragile_bones
					track = fragile_bones
					value = -50
				}
			}
		}
	}
}
