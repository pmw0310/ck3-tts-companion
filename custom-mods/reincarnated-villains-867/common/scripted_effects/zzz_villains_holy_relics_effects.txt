villains_create_and_enshrine_relic_effect = {
	set_artifact_rarity_$RARITY$ = yes
	$CREATOR$ = {
		save_scope_as = relic_owner
		create_artifact = {
			name = $NAME$
			description = $DESC$
			type = pedestal
			template = $TEMPLATE$
			visuals = $VISUALS$
			wealth = scope:wealth
			quality = scope:quality
			history = { type = created_before_history }
			modifier = artifact_placeholder_modifier
			save_scope_as = newly_created_artifact
		}
	}
	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = { name = villains_protected_holy_site_relic value = yes }
	}
	holy_site:$HOLY_SITE$ = {
		save_scope_as = relic_site
	}
	finalize_starting_relic_effect = yes
}
