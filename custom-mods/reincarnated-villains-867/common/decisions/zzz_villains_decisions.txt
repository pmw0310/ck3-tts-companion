		every_held_title = {

			limit = {
				tier = tier_duchy

				is_noble_family_title = no

			set_de_jure_liege_title = title:k_german_reich

		}

		every_held_title = {

			limit = {

				tier = tier_kingdom

				NOT = { this = title:k_german_reich }

			}

			every_in_de_jure_hierarchy = {

				limit = { tier = tier_duchy }

				set_de_jure_liege_title = title:k_german_reich

			}

		}

		title:k_german_reich = {

			set_de_jure_liege_title = title:e_germany

		}

			type = created

			save_scope_as = title_change

			add_claim_on_loss = no

		}

		title:k_german_reich = {

			change_title_holder = {

				holder = root

				change = scope:title_change

			}

		}

		resolve_title_and_vassal_change = scope:title_change

		set_primary_title_to = title:k_german_reich

		every_held_title = {

			limit = {

				tier = tier_kingdom

				NOT = { this = title:k_german_reich }

			}
