scripted_effect villains_kim_apply_canonization_core_effect = {
	if = {
		limit = {
			exists = scope:idolized_target
			exists = scope:paektu_consecration_province
			scope:idolized_target = {
				NOT = { has_trait = saint }
			}
		}
		custom_tooltip = kim_paektu_idolization_decision_effect_tt
		pam_canonize_saint_effect = {
			CANONIZER = root
			CANDIDATE = scope:idolized_target
			CONSECRATION_PROVINCE = scope:paektu_consecration_province
			PRESET = no
		}
	}
}

scripted_effect villains_amin_feudalize_tribal_counties_effect = {
	every_held_title = {
		limit = {
			tier = tier_county
			title_province = { has_holding_type = tribal_holding }
		}
		title_province = {
			set_holding_type = castle_holding
		}
	}
}

scripted_effect villains_rasputin_found_khlystism_core_effect = {
	faith:khlystism.main_rite = {
		set_can_convert_to_rite = yes
	}
	set_character_faith = faith:khlystism
	force_set_personal_tenet = tenet_polyamory
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	if = {
		limit = { NOT = { exists = title:d_khlystism.holder } }
		title:d_khlystism = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:title_change
	custom_tooltip = rasputin_found_khlystism_convert_followers_tt
	hidden_effect = {
		every_courtier_or_guest = {
			limit = { NOT = { faith = faith:khlystism } }
			set_character_faith = faith:khlystism
		}
		every_relation = {
			type = lover
			limit = {
				is_ruler = no
				NOT = { faith = faith:khlystism }
			}
			set_character_faith = faith:khlystism
		}
		if = {
			limit = { is_landed = yes }
			capital_county ?= {
				set_county_rite = rite:khlystism
			}
		}
	}
	add_character_modifier = rasputin_living_vessel_modifier
	dynasty ?= {
		add_dynasty_prestige = 500
	}
}
